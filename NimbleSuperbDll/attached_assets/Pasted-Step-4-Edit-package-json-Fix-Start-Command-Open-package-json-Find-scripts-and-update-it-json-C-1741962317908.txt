import { useState, useEffect } from "react";
import axios from "axios";

export default function Dashboard() {
  const [ticker, setTicker] = useState("AAPL");
  const [stockData, setStockData] = useState([]);
  const [newsSentiment, setNewsSentiment] = useState([]);

  const API_BASE_URL = "https://your-api-name.up.railway.app"; // Replace with your actual Railway API URL

  useEffect(() => {
    async function fetchData() {
      try {
        const stockResponse = await axios.get(`${API_BASE_URL}/stock/${ticker}`);
        const newsResponse = await axios.get(`${API_BASE_URL}/news/${ticker}`);
        setStockData(stockResponse.data);
        setNewsSentiment(newsResponse.data);
      } catch (error) {
        console.error("Error fetching data:", error);
      }
    }
    fetchData();
  }, [ticker]);

  return (
    <div className="p-6 bg-gray-900 min-h-screen text-white">
      <h1 className="text-3xl font-bold">📈 Real-Time Trading Dashboard</h1>
      
      <input
        className="p-2 mt-4 w-1/3 bg-gray-800 text-white rounded"
        type="text"
        value={ticker}
        onChange={(e) => setTicker(e.target.value)}
        placeholder="Enter stock symbol (e.g., TSLA, BTC-USD)"
      />

      {/* Stock Price Chart Placeholder */}
      <div className="mt-6">
        <h2 className="text-xl">Stock Price (Coming Soon...)</h2>
      </div>

      {/* News Sentiment Analysis */}
      <div className="mt-6">
        <h2 className="text-xl">News Sentiment Analysis</h2>
        <ul>
          {newsSentiment.map((news, index) => (
            <li key={index} className={`p-2 ${news.sentiment > 0 ? "text-green-500" : "text-red-500"}`}>
              {news.title} (Sentiment: {news.sentiment.toFixed(2)})
            </li>
          ))}
        </ul>
      </div>
    </div>
  );
}
